<script>
document.addEventListener("DOMContentLoaded", () => {
  const filterContainer = document.getElementById("photo-tag-filters");
  const resultPanel = document.getElementById("photo-tag-results");
  const resultTitle = document.getElementById("photo-tag-result-title");
  const resultList = document.getElementById("photo-tag-result-list");
  const cards = Array.from(document.querySelectorAll(".news-card[data-tags]"));
  const yearSections = Array.from(document.querySelectorAll(".main-content > details"));
  const cardTagButtons = Array.from(document.querySelectorAll(".tag-chip[data-tag]"));

  if (!filterContainer || !resultPanel || !resultList || !cards.length) return;

  const normalize = (value) => value.trim().toLowerCase();

  const tags = Array.from(
    new Set(
      cards.flatMap((card) =>
        (card.dataset.tags || "")
          .split(",")
          .map((tag) => tag.trim())
          .filter(Boolean)
      )
    )
  ).sort((a, b) => a.localeCompare(b));

  const tagMap = new Map(tags.map((tag) => [normalize(tag), tag]));
  const filterButtons = [];
  const activeFilters = new Set();

  const createButton = (label, filterValue) => {
    const button = document.createElement("button");
    button.type = "button";
    button.className = "tag-filter-button";
    button.textContent = label;
    button.dataset.filter = filterValue;
    filterContainer.appendChild(button);
    filterButtons.push(button);
    return button;
  };

  createButton("All", "all");
  tags.forEach((tag) => createButton(tag, tag));

  const selectedLabels = () =>
    Array.from(activeFilters)
      .map((tag) => tagMap.get(tag) || tag)
      .sort((a, b) => a.localeCompare(b));

  const toggleFilter = (rawValue) => {
    const value = normalize(rawValue || "");
    if (!value) return;

    if (value === "all") {
      activeFilters.clear();
    } else if (activeFilters.has(value)) {
      activeFilters.delete(value);
    } else {
      activeFilters.add(value);
    }

    applyFilter();
  };

  const renderResultList = (visibleCards) => {
    resultList.innerHTML = "";

    if (activeFilters.size === 0) {
      resultPanel.hidden = true;
      if (resultTitle) resultTitle.textContent = "Matching events";
      return;
    }

    const selected = selectedLabels();
    resultPanel.hidden = false;

    if (resultTitle) {
      resultTitle.textContent = `Matching events for: ${selected.join(", ")}`;
    }

    if (!visibleCards.length) {
      const li = document.createElement("li");
      li.textContent = "No matching events.";
      resultList.appendChild(li);
      return;
    }

    visibleCards.forEach((card) => {
      const li = document.createElement("li");
      const link = document.createElement("a");
      const title = card.dataset.title || "Untitled event";
      const meta = card.dataset.meta || "";
      link.href = `#${card.id}`;
      link.textContent = title;

      const metaSpan = document.createElement("span");
      metaSpan.className = "tag-result-meta";
      metaSpan.textContent = meta;

      li.appendChild(link);
      li.appendChild(metaSpan);
      resultList.appendChild(li);
    });
  };

  const updateActiveVisuals = () => {
    filterButtons.forEach((button) => {
      const value = normalize(button.dataset.filter || "");
      const isAll = value === "all";
      const isActive = isAll ? activeFilters.size === 0 : activeFilters.has(value);
      button.classList.toggle("is-active", isActive);
      button.setAttribute("aria-pressed", String(isActive));
    });

    cardTagButtons.forEach((button) => {
      const value = normalize(button.dataset.tag || "");
      const isActive = activeFilters.has(value);
      button.classList.toggle("is-active", isActive);
      button.setAttribute("aria-pressed", String(isActive));
    });
  };

  const applyFilter = () => {
    updateActiveVisuals();

    cards.forEach((card) => {
      const cardTags = (card.dataset.tags || "")
        .split(",")
        .map((tag) => normalize(tag))
        .filter(Boolean);

      const visible =
        activeFilters.size === 0 ||
        Array.from(activeFilters).every((tag) => cardTags.includes(tag));

      card.hidden = !visible;
    });

    yearSections.forEach((section) => {
      const visibleCards = section.querySelectorAll(".news-card[data-tags]:not([hidden])");
      section.hidden = visibleCards.length === 0;

      if (activeFilters.size === 0) {
        if (section.querySelector("summary strong")?.textContent?.trim() === "2025") {
          section.open = true;
        }
      } else {
        section.open = visibleCards.length > 0;
      }
    });

    const visibleCards = cards.filter((card) => !card.hidden);
    renderResultList(visibleCards);
  };

  filterContainer.addEventListener("click", (event) => {
    const button = event.target.closest(".tag-filter-button");
    if (!button) return;
    toggleFilter(button.dataset.filter || "");
  });

  document.addEventListener("click", (event) => {
    const tagButton = event.target.closest(".tag-chip[data-tag]");
    if (!tagButton) return;
    toggleFilter(tagButton.dataset.tag || "");

    const panelTop = filterContainer.closest(".tag-filter-panel");
    if (panelTop) {
      panelTop.scrollIntoView({ behavior: "smooth", block: "start" });
    }
  });

  applyFilter();
});
</script>
